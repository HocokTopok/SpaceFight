cmake_minimum_required(VERSION 3.28)
project(GalacticDominion VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(GD_BUILD_APP "Build SFML desktop game" ON)
option(GD_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
add_library(gd_core src/model.cpp src/game.cpp src/ai.cpp src/serialization.cpp)
target_include_directories(gd_core PUBLIC include)
target_compile_options(gd_core PRIVATE -Wall -Wextra -Wpedantic)
if(GD_SANITIZE)
  target_compile_options(gd_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(gd_core PUBLIC -fsanitize=address,undefined)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE gd_core)
  add_test(NAME core COMMAND core_tests)
  add_executable(game_tests tests/game_tests.cpp)
  target_link_libraries(game_tests PRIVATE gd_core)
  add_test(NAME gameplay COMMAND game_tests)
endif()
if(GD_BUILD_APP)
  include(FetchContent)
  set(SFML_BUILD_AUDIO OFF CACHE BOOL "" FORCE)
  set(SFML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(SFML_BUILD_TEST_SUITE OFF CACHE BOOL "" FORCE)
  set(SFML_BUILD_NETWORK ON CACHE BOOL "" FORCE)
  FetchContent_Declare(SFML GIT_REPOSITORY https://github.com/SFML/SFML.git GIT_TAG 3.0.2 GIT_SHALLOW ON)
  FetchContent_MakeAvailable(SFML)
  add_library(gd_network src/network.cpp)
  target_include_directories(gd_network PUBLIC include)
  target_link_libraries(gd_network PUBLIC gd_core SFML::Network)
  target_compile_options(gd_network PRIVATE -Wall -Wextra -Wpedantic)
  if(BUILD_TESTING)
    add_executable(network_tests tests/network_tests.cpp)
    target_link_libraries(network_tests PRIVATE gd_network)
    add_test(NAME network COMMAND network_tests)
  endif()
  add_executable(GalacticDominion src/app.cpp)
  if(APPLE)
    set_target_properties(GalacticDominion PROPERTIES
      MACOSX_BUNDLE TRUE
      MACOSX_BUNDLE_GUI_IDENTIFIER org.galacticdominion.game
      MACOSX_BUNDLE_BUNDLE_NAME "Galactic Dominion"
      MACOSX_BUNDLE_BUNDLE_VERSION "1.0.0"
      MACOSX_BUNDLE_SHORT_VERSION_STRING "1.0.0")
  endif()
  target_link_libraries(GalacticDominion PRIVATE gd_network SFML::Graphics)
  target_compile_definitions(GalacticDominion PRIVATE GD_ASSET_DIR="${CMAKE_CURRENT_SOURCE_DIR}/assets")
  target_compile_options(GalacticDominion PRIVATE -Wall -Wextra -Wpedantic)
  add_custom_command(TARGET GalacticDominion POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_CURRENT_SOURCE_DIR}/assets $<TARGET_FILE_DIR:GalacticDominion>/assets)
  install(TARGETS GalacticDominion BUNDLE DESTINATION . RUNTIME DESTINATION bin)
  if(NOT APPLE)
    install(DIRECTORY assets/ DESTINATION bin/assets)
  endif()
endif()
