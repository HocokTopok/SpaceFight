cmake_minimum_required(VERSION 3.28)
project(GalacticDominion VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(GD_BUILD_APP "Build SFML desktop game" ON)
option(GD_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
add_library(gd_core src/model.cpp)
target_include_directories(gd_core PUBLIC include)
target_compile_options(gd_core PRIVATE -Wall -Wextra -Wpedantic)
if(GD_SANITIZE)
  target_compile_options(gd_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(gd_core PUBLIC -fsanitize=address,undefined)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE gd_core)
  add_test(NAME core COMMAND core_tests)
endif()
if(GD_BUILD_APP)
  include(FetchContent)
  set(SFML_BUILD_AUDIO OFF CACHE BOOL "" FORCE)
  set(SFML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(SFML_BUILD_TEST_SUITE OFF CACHE BOOL "" FORCE)
  set(SFML_BUILD_NETWORK ON CACHE BOOL "" FORCE)
  FetchContent_Declare(SFML GIT_REPOSITORY https://github.com/SFML/SFML.git GIT_TAG 3.0.2 GIT_SHALLOW ON)
  FetchContent_MakeAvailable(SFML)
endif()
